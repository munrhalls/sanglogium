## 3. Validation Rules - Contract Based

### Required Field Validation
```javascript
const RequiredFields = [
  { field: 'name', type: 'string', minLength: 1, maxLength: 200 },
  { field: 'displayPrice', type: 'number', min: 0.01, max: 10000 },
  { field: 'sku', type: 'string', minLength: 3, maxLength: 50 },
  { field: 'brand', type: 'reference', to: 'brand' },
  { field: 'image', type: 'image' }
];

const validateRequired = (data) => {
  const errors = [];
  
  RequiredFields.forEach(({ field, type, ...rules }) => {
    const value = data[field];
    
    if (!value) {
      errors.push(`${field}: Required field missing`);
      return;
    }
    
    if (type === 'string' && typeof value !== 'string') {
      errors.push(`${field}: Must be string, got ${typeof value}`);
    }
    
    if (type === 'number' && typeof value !== 'number') {
      errors.push(`${field}: Must be number, got ${typeof value}`);
    }
    
    if (type === 'reference' && (!value._ref || value._type !== 'reference')) {
      errors.push(`${field}: Must be valid reference`);
    }
    
    if (type === 'image' && value._type !== 'image') {
      errors.push(`${field}: Must be valid image object`);
    }
  });
  
  return errors;
};
```

### Schema Shape Validation
```javascript
const validateShape = (data) => {
  const errors = [];
  
  // Check no extra fields
  const allowedFields = Object.keys(FieldMapping.scraped);
  const presentFields = Object.keys(data);
  
  presentFields.forEach(field => {
    if (!allowedFields.includes(field)) {
      errors.push(`${field}: Extra field not allowed`);
    }
  });
  
  // Check array structures
  if (data.gallery && !Array.isArray(data.gallery)) {
    errors.push('gallery: Must be array');
  }
  
  if (data.overviewFields && !Array.isArray(data.overviewFields)) {
    errors.push('overviewFields: Must be array');
  }
  
  if (data.specifications && !Array.isArray(data.specifications)) {
    errors.push('specifications: Must be array');
  }
  
  return errors;
};
```

## 5. Containment Against Drift

### Contract Locking
```javascript
// Freeze the contract - no changes allowed
const SchemaContract = Object.freeze({
  requiredFields: Object.freeze(RequiredFields),
  fieldMapping: Object.freeze(FieldMapping),
  validationRules: Object.freeze({
    required: validateRequired,
    shape: validateShape
  })
});

// Any deviation throws error
const enforceContract = (data) => {
  const errors = [
    ...SchemaContract.validationRules.required(data),
    ...SchemaContract.validationRules.shape(data)
  ];
  
  if (errors.length > 0) {
    throw new Error(`Contract violation: ${errors.join(', ')}`);
  }
  
  return data;
};
```
